```
<?xml version="1.0" encoding="ISO-8859-15" standalone="no"?>
<Anmeldungssteuern xmlns="http://finkonsens.de/elster/elsteranmeldung/ustva/v{YEAR}" version="{YEAR}">
  <Erstellungsdatum>YYYYMMDD</Erstellungsdatum>
  <Steuerfall>
    <Umsatzsteuervoranmeldung>
      <Jahr>{report.Jahr}</Jahr>
      <Zeitraum>{report.Zeitraum}</Zeitraum>
      <Steuernummer>{converted 13-digit}</Steuernummer>
      <!-- ONLY the Kz present in report.Lines: base = integer euros, tax = 2dp with '.' separator -->
      <Kz81>...</Kz81> <Kz86>...</Kz86> <Kz41>...</Kz41> <Kz83>...</Kz83>
    </Umsatzsteuervoranmeldung>
  </Steuerfall>
</Anmeldungssteuern>
```
